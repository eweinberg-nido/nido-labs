function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    var rows = [];
    if (data.length > 1) {
      var headers = data[0];
      for (var i = 1; i < data.length; i++) {
        var row = {};
        for (var j = 0; j < headers.length; j++) {
          row[headers[j]] = data[i][j];
        }
        rows.push(row);
      }
    }
    var json = JSON.stringify(rows);
    var callback = e && e.parameter ? e.parameter.callback : null;
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + json + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(json)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    var errObj = JSON.stringify({ error: err.toString() });
    var callback = e && e.parameter ? e.parameter.callback : null;
    if (callback) {
      return ContentService.createTextOutput(callback + '(' + errObj + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(errObj)
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Booking ID", "Block ID", "Date", "Time", "Block", "Teacher", 
        "Course", "Phase", "Topic", "Attendance", "Mentor Name", "Mentor Email", 
        "Organization", "Notes", "Submitted At"
      ]);
    }
    var body = e.postData ? e.postData.contents : "";
    var entries = JSON.parse(body);
    if (!Array.isArray(entries)) entries = [entries];
    entries.forEach(function(item) {
      sheet.appendRow([
        item.bookingId || "",
        item.slotId || item.blockId || "",
        item.date || "",
        item.time || "",
        item.block || "",
        item.teacher || "",
        item.course || "",
        item.phase || "",
        item.topic || "",
        item.attendance || "In Person",
        item.mentorName || "",
        item.mentorEmail || "",
        item.mentorOrg || "",
        item.mentorNotes || "",
        item.createdAt || new Date().toISOString()
      ]);
    });
    return ContentService.createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
